def create_face_mask(shape):
    \"\"\"Create oval mask to suppress edges\"\"\"
    h, w = shape
    y, x = np.ogrid[:h, :w]
    center_y, center_x = h // 2, w // 2
    
    # Oval mask (wider in horizontal)
    mask = ((x - center_x)**2 / (w * 0.4)**2 + 
            (y - center_y)**2 / (h * 0.45)**2) <= 1
    
    # Smooth edges with Gaussian
    from scipy.ndimage import gaussian_filter
    mask = gaussian_filter(mask.astype(float), sigma=5)
    return mask
